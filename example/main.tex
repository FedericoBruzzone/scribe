% =====================================================================
% Layout selector. By default this file builds the two-column layout
% with a full-width abstract. The repo-root script ../regen-docs.sh
% overrides it on the command line to render all three preview variants
% from this single source, e.g.:
%   pdflatex "\def\scribelayout{onecolumn}\input{main.tex}"
% Accepted values:
%   onecolumn          - single column
%   twocolumn          - two columns, abstract in the first column
%   twocolumnabstract  - two columns, abstract spanning the top (default)
% =====================================================================
\providecommand{\scribelayout}{twocolumnabstract}
\def\scribeonecol{onecolumn}
\def\scribetca{twocolumnabstract}

\ifx\scribelayout\scribeonecol
    \documentclass[lineno, letterpaper]{scribe}            % single column
\else
    \documentclass[lineno, letterpaper, twocolumn]{scribe} % two columns
\fi
